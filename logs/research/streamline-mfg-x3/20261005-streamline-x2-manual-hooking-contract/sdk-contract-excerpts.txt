123: #### 4.1 DirectX
124: 
125: There are two scenarios depending on whether `sl.interposer.lib` is linked directly or not with the host application:
126: 
127: ```cpp
128: 
129: //! SL LIB LINKED WITH THE GAME, SL PROXIES ARE PROVIDED TO THE HOST
130: 
131: if(!sl::security::verifyEmbeddedSignature(PATH_TO_SL_IN_YOUR_BUILD + "/sl.interposer.dll"))
132: {
133:     // SL module not signed, disable SL
134: }
135: else
136: {
137:     // SL digitally signed, OK to use it!
138: 
139:     // D3D11
140:     // 
141:     // IMPORTANT: Note that for D3D11 there is NO proxy for a device in any scenario
142:     ID3D11Device* nativeDevice;
143:     [nativeDevice, proxySwapchain] = D3D11CreateDeviceAndSwapChain();
144: 
145:     // D3D12
146:     proxyDevice = D3D12CreateDevice();
147: 
148:     ID3D12Device* nativeDevice{};
149:     if(SL_FAILED(result, slGetNativeInterface(proxyDevice, &nativeDevice))
150:     {
151:         // Handle error, check logs
152:     }
153:     // Normally this is not done for D3D11 (see above)
154:     auto proxyFactory = DXGICreateFactory();
155:     auto proxySwapChain = proxyFactory->CreateSwapChain();
156: 
157:     // DXGI
158:     //
159:     // This part is identical for D3D11 and D3D12    
160:     IDXGISwapChain* nativeSwapchain{};
161:     if(SL_FAILED(result, slGetNativeInterface(proxySwapChain, &nativeSwapchain)))
162:     {
163:         // Handle error, check logs
164:     }
165: }
166: ```
167: 
168: ```cpp
169: 
170: //! SL LIB NOT LINKED WITH THE GAME, NATIVE INTERFACES ARE PROVIDED TO THE HOST
171: 
172: if(!sl::security::verifyEmbeddedSignature(PATH_TO_SL_IN_YOUR_BUILD + "/sl.interposer.dll"))
173: {
174:     // SL module not signed, disable SL
175: }
176: else
177: {
178:     // SL digitally signed, we can load and use it!
179:     auto mod = LoadLibrary(PATH_TO_SL_IN_YOUR_BUILD + "/sl.interposer.dll");
180: 
181:     // Declare all SL functions here, showing only the one we are using
182:     SL_FUN_DECL(slUpgradeInterface);
183:     SL_FUN_DECL(slGetNativeInterface);
184: 
185:     // Get all SL functions here, showing only the one we are using
186:     auto slUpgradeInterface = reinterpret_cast<PFunSlGetUpgradeInterface>(GetProcAddress(mod, "slUpgradeInterface"));
187:     auto slGetNativeInterface = reinterpret_cast<PFunSlGetNativeInterface>(GetProcAddress(mod, "slGetNativeInterface"));
188: 
189:     // D3D11
190:     // 
191:     // IMPORTANT: Note that for D3D11 there is NO proxy for a device in any scenario
192:     ID3D11Device* nativeDevice;
193:     [nativeDevice, nativeSwapchain] = D3D11CreateDeviceAndSwapChain();
194: 
195:     // D3D12
196:     auto nativeDevice = D3D12CreateDevice();
197:     auto nativeFactory = DXGICreateFactory();
198: 
199:     ID3D12Device* proxyDevice = nativeDevice;
200:     if(SL_FAILED(result, slUpgradeInterface(&proxyDevice))
201:     {
202:         // Handle error, check logs
203:     }
204: 
205:     //! IMPORTANT: Any create swap chain API must be intercepted as specified by `enum class FunctionHookID` in sl_hooks.h
206:     //!
207:     //! Therefore here we are using proxy factory and not a native one!
208:     IDXGIFactory* proxyFactory = nativeFactory
209:     if(SL_FAILED(result, slUpgradeInterface(&proxyFactory)))
210:     {
211:         // Handle error, check logs
212:     }
213: 
214:     //! Now we can obtain our proxy swap-chain
215:     proxySwapChain = proxyFactory->CreateSwapChain();
216: 
217:     //! Next we get our proxy device
218:     ID3D12Device* proxyDevice = nativeDevice;
219:     if(SL_FAILED(result, slUpgradeInterface(&proxyDevice)))
220:     {
221:         // Handle error, check logs
222:     }
223: 
224:     //! Native swap-chain so we can call non-intercepted API directly
225:     IDXGISwapChain* nativeSwapchain{};
226:     if(SL_FAILED(result, slGetNativeInterface(proxySwapChain, &nativeSwapchain)))
227:     {
228:         // Handle error, check logs
229:     }
230: }
231: ```

233: > **IMPORTANT**
234: > One should use native interfaces EVERYWHERE in the host application EXCEPT for the APIs which are hooked by SL (listed as enums in sl_hooks.h).

320: // Always secure load SL modules
321: if(!sl::security::verifyEmbeddedSignature(PATH_TO_SL_IN_YOUR_BUILD + "/sl.interposer.dll"))

323:     // SL module not signed, disable SL
324: }

327:     auto mod = LoadLibray(PATH_TO_SL_IN_YOUR_BUILD + "/sl.interposer.dll");
