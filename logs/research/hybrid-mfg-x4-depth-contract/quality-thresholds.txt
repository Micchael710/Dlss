CLASSIFICATION RULES:
  STRUCTURAL_VALIDATION:
    PASS if:
      - G25, G50, G75 generated
      - Changed from sentinels
      - Resources distinct
      - Frames distinct from endpoints
      - G25, G50, G75 distinct from each other
      - Strict temporal ordering: A < G25 < G50 < G75 < B
      - D3D12_VALIDATION_ERRORS == 0
      - DEVICE_LOST=false, DEVICE_REMOVED=false, CRASH=false, TIMEOUT=false

  CUSTOM_INTERPOLATOR_QUALITY:
    PASS if G25 & G75 satisfy both overall and challenging limits.
    FAIL_OCCLUSION_REGIONS if overall passes but challenging fails.
    FAIL_GLOBAL_REPROJECTION if overall fails.

  NVIDIA_G50_QUALITY:
    PASS if G50 satisfies overall and challenging limits.
    FAIL_CHALLENGING_REGIONS if overall passes but challenging fails.
    FAIL_GLOBAL if overall fails.

  QUALITY_VALIDATION:
    PASS if both CUSTOM_INTERPOLATOR_QUALITY == PASS and NVIDIA_G50_QUALITY == PASS, else FAIL.
